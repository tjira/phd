\makeatletter\let\percentchar\@percentchar\makeatother

\newcommand{\filetotitle}{
    \directlua{
        local name = status.filename:match("([^/]+).tex$"):gsub("_", " "):gsub("^\percentchar d+-", "")

        local exceptions = {["and"] = true, ["of"] = true, ["the"] = true, ["in"] = true, ["with"] = true, ["vs"] = true}

        local words = {}; local first = true

        for word in name:gmatch("\percentchar S+") do

            local word = word:lower()

            if is_first or not exceptions[word] then
                word = word:sub(1,1):upper() .. word:sub(2)
            end

            table.insert(words, word); first = false
        end

        tex.sprint(table.concat(words, " "))
    }
}

\newcommand{\includechapters}{
    \directlua{
        local folders = {}

        for folder in require("lfs").dir("chapters") do
            if folder ~= "." and folder ~= ".." then
                table.insert(folders, folder)
            end
        end

        table.sort(folders)

        for _, folder in ipairs(folders) do
            tex.sprint("\\include{chapters/" .. folder .. "/main" .. "}")
        end
    }
}

\newcommand{\inputfolder}[1]{
    \directlua{
        local files = {}

        for file in require("lfs").dir("#1") do
            if file:sub(-4) == ".tex" then
                table.insert(files, file)
            end
        end

        table.sort(files)

        for _, file in ipairs(files) do
            tex.sprint("\\input{" .. "#1" .. "/" .. file .. "}")
        end
    }
}
