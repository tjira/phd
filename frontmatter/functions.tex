\begin{luacode*}
    function filetotitle()
        tex.sprint((status.filename:match("([^/]+).tex$"):gsub("_", " "):gsub("^%d+-", "")))
    end
\end{luacode*}

\begin{luacode*}
    function foldertotitle()
        tex.sprint((status.filename:match("([^/]+)/[^/]+.tex$"):gsub("_", " "):gsub("^%d+-", "")))
    end
\end{luacode*}

\begin{luacode*}
    function includechapters(path)
        local folders = {}

        for folder in require("lfs").dir(path) do
            if folder ~= "." and folder ~= ".." then
                table.insert(folders, folder)
            end
        end

        table.sort(folders)

        for _, folder in ipairs(folders) do

            local not_include = string.find(folder, "Papers") and os.getenv("PAPERS") ~= "1"

            if not not_include then
                tex.sprint("\\include{" .. path .. "/" .. folder .. "/main" .. "}")
            end
        end
    end
\end{luacode*}

\begin{luacode*}
    function inputfolder(path)
        local current_dir = status.filename:match("(.*)/") or "."

        path = current_dir .. "/" .. path; local files = {}

        for file in require("lfs").dir(path) do
            if file:sub(-4) == ".tex" then
                table.insert(files, file)
            end
        end

        table.sort(files)

        for _, file in ipairs(files) do
            tex.sprint("\\input{" .. path .. "/" .. file .. "}")
        end
    end
\end{luacode*}
