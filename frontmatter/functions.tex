\begin{luacode*}
    function filedirpath()
        local path = status.filename:match("(.*)/")

        tex.sprint(path or ".")
    end
\end{luacode*}

\begin{luacode*}
    function filetotitle()
        tex.sprint((status.filename:match("([^/]+)%.tex$"):gsub("_", " "):gsub("^%d+-", "")))
    end
\end{luacode*}

\begin{luacode*}
    function foldertotitle()
        tex.sprint((status.filename:match("([^/]+)/[^/]+%.tex$"):gsub("_", " "):gsub("^%d+-", "")))
    end
\end{luacode*}

\begin{luacode*}
    function includechapters(path)
        local folders = {}; local papers = os.getenv("PAPERS") == "1"; local chapter = os.getenv("CHAPTER")

        for folder in require("lfs").dir(path) do

            local include = not folder:find("Paper") or papers

            if chapter and chapter ~= "" then
                include = include and (folder == chapter)
            end

            if folder:sub(1, 1) ~= "." and include then
                table.insert(folders, folder)
            end
        end

        table.sort(folders)

        for _, folder in ipairs(folders) do
            tex.sprint(string.format("\\include{%s/%s/main}", path, folder))
        end
    end
\end{luacode*}

\begin{luacode*}
    function inputfolder(path)
        path = status.filename:match("(.*)/") .. "/" .. path; local files = {}

        for file in require("lfs").dir(path) do

            local input = file:match("%.tex$")

            if input then
                table.insert(files, file)
            end
        end

        table.sort(files)

        for _, file in ipairs(files) do
            tex.sprint(string.format("\\input{%s/%s}", path, file))
        end
    end
\end{luacode*}
